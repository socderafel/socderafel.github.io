public static int sumav(int[] v, int i, int x) {
int suma = 0;
if (i<0) return suma;
if (v[i]==x) return suma;
for (int j=0; j<=i; j++) {
suma = suma + v[j];
}
System.out.println("Suma parcial: " + suma);
return suma + sumav(v, i-1, x);
