import java.util.ArrayList;

public class P714FaseFinal {

	public static void main(String[] args) {

		Scanner sc = new Scanner(System.in);

		int numero; 

		do {			

			numero = sc.nextInt();

			

			if(numero != 0) {

				int nEquipos = numero;

				String ganador = "";

	

				// Rellenar la fase actual con equipos

				ArrayList<String> faseActual = new ArrayList<String>();

				for(int i = 0; i < nEquipos; i++) {

					faseActual.add(sc.next());

				}

	

				// Mientras haya equipos evaluar resultados y obtener ganador

				while(nEquipos > 0) {

					// Lista de equipos que pasan a la siguiente fase

					ArrayList<String> faseSiguiente = new ArrayList<String>();

					//System.out.println(fase);

	

					// Obtener los resultados de cada fase

					for(int i = 0; i < nEquipos/2; i++) {

						int golesEquipo1 = sc.nextInt();

						int golesEquipo2 = sc.nextInt();

						//System.out.println(golesEquipo1 + " " + golesEquipo2);

	

						// El equipo que gana pasa a la fase siguiente

						if(golesEquipo1 > golesEquipo2) {

							faseSiguiente.add(faseActual.get(0));

							ganador = faseActual.get(0);

						} else {

							faseSiguiente.add(faseActual.get(1));

							ganador = faseActual.get(1);

						} 

						// Elimino los dos primeros equipos de la fase actual

						faseActual.remove(0);

						faseActual.remove(0);

					}

					

					// Actualizar el número de equipos de la siguiente fase

					nEquipos /= 2; 

					

					// La fase siguiente ahora es la actual

					faseActual = faseSiguiente;

				}

	

				// Mostrar ganador

				System.out.println(ganador);

			}

		}while(numero != 0); // Mientras haya equipos...

	}

}

		Scanner sc = new Scanner(System.in);

		int nEventos = sc.nextInt(); // Obtener número de casos

		

		int[] listaTiempos = new int[nEventos];

		for(int i = 0; i < nEventos; i++) {

			listaTiempos[i] = sc.nextInt();

			sc.nextLine();

		}	

		

		int nConsultas = sc.nextInt();

		int elegido = 0;

		for(int i = 0; i < nConsultas; i++) {

			int nEventosConsulta = sc.nextInt();

			

		    int min = 100000000;

		    int rangoMinimo = -1;

			for(int j = 0; j < nEventos-nEventosConsulta; j++) {

				System.out.println(nEventosConsulta);

				int tiempo1 = listaTiempos[j];

				HashMap<Integer, Integer> contadorEventos = new HashMap<Integer, Integer>();

				for(int k = j; k < nEventosConsulta+j; k++) {

					int tiempo2 = listaTiempos[k];

				    int periodo = tiempo2 - tiempo1;

				    System.out.println(tiempo1 + " " + tiempo2);

				    if(periodo < min) {

				    	min = periodo;

				    	rangoMinimo = j;

				    	contadorEventos.put(j, contadorEventos.getOrDefault(tiempo1,0) + 1);

				    }

				}

				

				int maximo = 0;

				for(Integer posicion : contadorEventos.keySet()) {

					if(contadorEventos.get(posicion) > maximo) {

						maximo = contadorEventos.get(posicion);

						elegido = posicion;

					}

				}

			}

			

			

			System.out.println(listaTiempos[elegido] + " " + listaTiempos[elegido + nEventosConsulta]);

		

		}

	}

}

	public static void main(String[] args) {

		Scanner sc = new Scanner(System.in);

		int casos = sc.nextInt(); // Obtener número de casos

		

		for(int i = 0; i < casos; i++) {

			String numero = sc.next();

			

			int codigoVerificacion = 0;

			for(int j = 0; j < numero.length(); j++) {

				int digito = Integer.parseInt(numero.charAt(j) + "");

				if((j+1) % 2 != 0) { // Impar

					// Multiplicar por 3

					digito = digito * 3;

					

					// Obtener menor dígito a la derecha y sumar a digito

					int min = 10;

					for (int k = j+1; k < numero.length(); k++) {

						min = Math.min(min, Integer.parseInt(numero.charAt(k) + ""));

					}

					digito += min;

					

					// Actualizar código de verificación

					codigoVerificacion += digito;

					

				} else { // Par

					// Multiplicar por 3

					digito = digito * 2;

					// Obtener menor dígito a la derecha y sumar a digito

					int max = -1;

					for (int k = j-1; k >= 0; k--) {

						max = Math.max(max, Integer.parseInt(numero.charAt(k) + ""));

					}

					digito += max;

					// Actualizar código de verificación

					codigoVerificacion += digito;

				}

			}

			// Mostrar código verificación

			System.out.println(codigoVerificacion);

		}

	}

import java.util.ArrayList;

import java.util.Collection;

import java.util.Collections;

import java.util.HashMap;

import java.util.Map;

import java.util.Scanner;

public class P709SedeMundial {

	public static void main(String[] args) {

		Scanner sc = new Scanner(System.in);

		int casos;

		do {

			casos = sc.nextInt();

			if(casos == 0)

				break;

			

			// Insertar los países y candidaturas

			Map<String, Integer> paisesCandidaturas = new HashMap<String, Integer>();

			for(int i = 0; i < casos; i++) {

				String pais = sc.next();

				if(paisesCandidaturas.containsKey(pais))

					paisesCandidaturas.put(pais, paisesCandidaturas.get(pais) + 1);

				else

					paisesCandidaturas.put(pais, 1);

			}

			

			// Obtener el máximo

			int max = 0;

			for(String pais : paisesCandidaturas.keySet()) {

				max = Math.max(max, paisesCandidaturas.get(pais));

			}

			// Comprobar si hay empate

			int contador = 0;

			String paisGanador = "";

			for(String pais : paisesCandidaturas.keySet()) {

				if (paisesCandidaturas.get(pais) == max) {

					paisGanador = pais;

					contador++;		

				}

			}

			

			// Mostrar resultado

			if (contador > 1) {

				System.out.println("EMPATE");

			} else {

				System.out.println(paisGanador);

			}

			

		}while(casos != 0);	

	}

	

}
