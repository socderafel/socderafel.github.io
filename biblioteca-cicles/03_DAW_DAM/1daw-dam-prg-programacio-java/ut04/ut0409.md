# 4.9 InternacionalizacionFX

#### 📦 InternacionalizacionFXController.java

```java
package application;

import javafx.event.ActionEvent;
import javafx.fxml.FXML;
import javafx.scene.control.Button;
import javafx.scene.control.Label;

import java.util.Locale;
import java.util.ResourceBundle;

public class InternacionalizacionFXController {
    @FXML
    private Button btnEspanyol;

    @FXML
    private Button btnIngles;

    @FXML
    private Label label;

    @FXML
    private Label label2;

    @FXML
    public void initialize() {
    }

    @FXML
    void clickBotonEspanyol(ActionEvent event) {
        PrincipalInternacionalizacionFX.setLocale(new Locale("es", "ES"));

        ResourceBundle bundle = PrincipalInternacionalizacionFX.getBundle();

        label.setText(bundle.getString("label"));
        label2.setText(bundle.getString("label2"));
        btnEspanyol.setText(bundle.getString("button.espanyol"));
        btnIngles.setText(bundle.getString("button.ingles"));       
    }

    @FXML
    void clickBotonIngles(ActionEvent event) {
        PrincipalInternacionalizacionFX.setLocale(new Locale("en", "UK"));
       
        ResourceBundle bundle = PrincipalInternacionalizacionFX.getBundle();
        
        label.setText(bundle.getString("label"));
        label2.setText(bundle.getString("label2"));
        btnEspanyol.setText(bundle.getString("button.espanyol"));
        btnIngles.setText(bundle.getString("button.ingles"));    
    }
}
```

#### 📦 PrincipalInternacionalizacionFX.java

```java
package application;
	
import java.util.Locale;
import java.util.ResourceBundle;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Scene;
import javafx.scene.layout.AnchorPane;
import javafx.stage.Stage;

public class PrincipalInternacionalizacionFX extends Application {
	
	private static Locale currentLocale = new Locale("es", "ES");
    private static ResourceBundle bundle = ResourceBundle.getBundle("recursos.traduccion", currentLocale);
    
    public static void setLocale(Locale locale) {
	     currentLocale = locale;
	     bundle = ResourceBundle.getBundle("recursos.traduccion", locale);
	}
    
    public static ResourceBundle getBundle() {
        return bundle;
    }
    
	@Override
	public void start(Stage primaryStage) {
		try {
			//Locale locale = new Locale("en","UK"); traduccion_en_UK.properties
			
			//ResourceBundle bundle = ResourceBundle.getBundle("recursos.traduccion", locale);
			
			AnchorPane root = (AnchorPane)FXMLLoader.load(getClass().getResource("InternacionalizacionFX.fxml"),bundle);
			Scene scene = new Scene(root);
			scene.getStylesheets().add(getClass().getResource("application.css").toExternalForm());
			primaryStage.setScene(scene);
			primaryStage.show();
		} catch(Exception e) {
			e.printStackTrace();
		}
	}
	
	public static void main(String[] args) {
		launch(args);
	}
}
```
